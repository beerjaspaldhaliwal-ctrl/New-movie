  end if

      set_control_par(%ui_id[$ui_i], $CONTROL_PAR_HIDE, $HIDE_PART_NOTHING)
    else
      set_control_par(%ui_id[$ui_i], $CONTROL_PAR_HIDE, $HIDE_WHOLE_CONTROL)

function play_voice
  $pv_id := play_note($pv_note, 127, $pv_off, 0)
  set_event_par_arr($pv_id, $EVENT_PAR_ALLOW_GROUP, 0, $ALL_GROUPS)
  set_event_par_arr($pv_id, $EVENT_PAR_ALLOW_GROUP, 1, $pv_grp)
  change_vol($pv_id, $pv_vol, 0)
  change_tune($pv_id, $pv_tune + $bend, 0)
  if ($pv_pan # 0)
    change_pan($pv_id, $pv_pan, 0)
  end if
  if ($pv_fade > 0)
    fade_in($pv_id, $pv_fade)
  end if
  if ($nv_n < 32)
    %nv_id[$nv_n] := $pv_id
    %nv_r[$nv_n] := $pv_role
    %nv_b[$nv_n] := $pv_vol
    inc($nv_n)
  end if
end function

function reg_new
  $rn_i := 0
  while ($rn_i < $nv_n)
    if (%src_n[$rs_src] < $MAXV)
      $rn_j := $rs_src * $MAXV + %src_n[$rs_src]
      %src_v[$rn_j] := %nv_id[$rn_i]
      %src_r[$rn_j] := %nv_r[$rn_i]
      %src_b[$rn_j] := %nv_b[$rn_i]
      %src_n[$rs_src] := %src_n[$rs_src] + 1
    else
      fade_out(%nv_id[$rn_i], 30000, 1)
    end if
    inc($rn_i)
  end while
end function

function release_src
  $rl_i := 0
  while ($rl_i < %src_n[$rs_src])
    fade_out(%src_v[$rs_src * $MAXV + $rl_i], $rs_time, 1)
    inc($rl_i)
  end while
  %src_n[$rs_src] := 0
  %src_sus[$rs_src] := 0
end function

function pend_new
  $pn_i := 0
  while ($pn_i < $nv_n)
    if ($pq_n < 512)
      %pq_id[$pq_n] := %nv_id[$pn_i]
      %pq_t[$pq_n] := $pq_rel
      inc($pq_n)
    else
      fade_out(%nv_id[$pn_i], 20000, 1)
    end if
    inc($pn_i)
  end while
end function

function process_pending
  $pp_i := 0
  while ($pp_i < $pq_n)
    if (%pq_t[$pp_i] <= $tick)
      fade_out(%pq_id[$pp_i], $rel_us, 1)
      dec($pq_n)
      %pq_id[$pp_i] := %pq_id[$pq_n]
      %pq_t[$pp_i] := %pq_t[$pq_n]
    else
      inc($pp_i)
    end if
  end while
end function

function release_all
  $ra_s := 0
  while ($ra_s < 129)
    if (%src_n[$ra_s] > 0)
      $rs_src := $ra_s
      call release_src
    end if
    %src_sus[$ra_s] := 0
    inc($ra_s)
  end while
  $ra_i := 0
  while ($ra_i < $pq_n)
    fade_out(%pq_id[$ra_i], $rs_time, 1)
    inc($ra_i)
  end while
  $pq_n := 0
end function

function pedal_up
  $pu_s := 0
  while ($pu_s < 128)
    if (%src_sus[$pu_s] = 1 and %src_held[$pu_s] = 0)
      $rs_src := $pu_s
      $rs_time := $rel_us
      call release_src
    end if
    inc($pu_s)
  end while
end function

function apply_cutoff
  $fx_v := $cut_base + $cut_mod + $cut_lfo
  if ($fx_v < 0)
    $fx_v := 0
  end if
  if ($fx_v > 1000000)
    $fx_v := 1000000
  end if
  set_engine_par($ENGINE_PAR_CUTOFF, $fx_v, -1, 0, $NI_INSERT_BUS)
end function

function vel_curve
  ~vc_r := real($vc_in) / 127.0
  ~vc_r := pow(~vc_r, pow(2.0, real(0 - $vcurve) / 50.0))
  $vc_curved := int(~vc_r * 127.0)
  if ($vc_curved < 1)
    $vc_curved := 1
  end if
  $vc_db := int(real($vrange) * 1000.0 * (~vc_r - 1.0))
end function

function calc_dyn
  ~fr := real($dyn) / 127.0
  if (~fr < 0.001)
    ~fr := 0.001
  end if
  $dyn_db := int(20000.0 * log10(~fr) * 1.2)
end function

function update_dyn
  $ud_s := 0
  while ($ud_s < 128)
    if (%src_n[$ud_s] > 0)
      $ud_i := 0
      while ($ud_i < %src_n[$ud_s])
        $ud_j := $ud_s * $MAXV + $ud_i
        change_vol(%src_v[$ud_j], %src_b[$ud_j] + $dyn_db, 0)
        inc($ud_i)
      end while
    end if
    inc($ud_s)
  end while
end function

function drum_hit
  $d_pc := %key_pc[$dh_note]
  if ($d_pc < 0)
    exit
  end if
  if (%pz_cut[$d_pc] > 0)
    $d_i := 0
    while ($d_i < $NUM_PC)
      if (%pz_cg[$d_i] = %pz_cut[$d_pc] and $d_i # $d_pc)
        $d_j := 0
        while ($d_j < 8)
          if (%pv_ring[$d_i * 8 + $d_j] # 0)
            fade_out(%pv_ring[$d_i * 8 + $d_j], 45000, 1)
            %pv_ring[$d_i * 8 + $d_j] := 0
          end if
          inc($d_j)
        end while
      end if
      inc($d_i)
    end while
  end if
  $vc_in := $dh_vel
  call vel_curve
  $d_l := 0
  while ($d_l < %pz_nl[$d_pc] - 1 and $vc_curved > %pz_split[$d_pc * 4 + $d_l])
    inc($d_l)
  end while
  $d_r := %pz_rr[$d_pc * 4 + $d_l]
  if (%pz_rrn[$d_pc * 4 + $d_l] > 1)
    %pz_rr[$d_pc * 4 + $d_l] := ($d_r + 1 + random(0, %pz_rrn[$d_pc * 4 + $d_l] - 2)) mod %pz_rrn[$d_pc * 4 + $d_l]
  end if
  $d_g := %grp[%pz_g[($d_pc * 4 + $d_l) * 4 + $d_r]]
  if ($d_g = $NI_NOT_FOUND)
    exit
  end if
  $nv_n := 0
  $pv_note := $dh_note
  $pv_grp := $d_g
  $pv_vol := $vc_db + %bus_db[%pz_bus[$d_pc]] + random(0 - $hum * 15, $hum * 15)
  $pv_tune := $tune * 1000 + random(0 - $hum * 40, $hum * 40)
  $pv_pan := (%pz_pan[$d_pc] * $spread) / 100
  $pv_off := 0
  $pv_fade := 0
  $pv_role := 3
  if (%pz_hold[$d_pc] = 1)
    $pv_vol := $pv_vol + $dyn_db
  end if
  call play_voice
  %pv_ring[$d_pc * 8 + %pv_pos[$d_pc]] := $pv_id
  %pv_pos[$d_pc] := (%pv_pos[$d_pc] + 1) mod 8
  if (%pz_hold[$d_pc] = 1)
    %nv_b[0] := $pv_vol - $dyn_db
    $rs_src := $dh_note
    call reg_new
  end if
end function

on note
  ignore_event($EVENT_ID)
  if (%key_pc[$EVENT_NOTE] < 0)
    exit
  end if
  inc($keys_down)
  %src_held[$EVENT_NOTE] := 1
  $dh_note := $EVENT_NOTE
  $dh_vel := $EVENT_VELOCITY
  call drum_hit
  if ($repeat > 0 and %pz_hold[%key_pc[$EVENT_NOTE]] = 0)
    if (%rep_vel[$EVENT_NOTE] = 0)
      inc($rep_n)
    end if
    %rep_vel[$EVENT_NOTE] := $EVENT_VELOCITY
    %rep_next[$EVENT_NOTE] := $tick + %rep_ticks[$repeat]
  end if
end on

on release
  if (%key_pc[$EVENT_NOTE] >= 0)
    if (%rep_vel[$EVENT_NOTE] > 0)
      %rep_vel[$EVENT_NOTE] := 0
      dec($rep_n)
    end if
    dec($keys_down)
    if ($keys_down < 0)
      $keys_down := 0
    end if
    %src_held[$EVENT_NOTE] := 0
    if (%src_n[$EVENT_NOTE] > 0)
      $rs_src := $EVENT_NOTE
      $rs_time := $rel_us
      call release_src
    end if
  end if
end on

on controller
  if ($CC_NUM = 64 and $use_pedal = 1)
    if (%CC[64] >= 64)
      $pedal := 1
    else
      if ($pedal = 1)
        $pedal := 0
        call pedal_up
      end if
    end if
  end if
  if ($CC_NUM = 120 or $CC_NUM = 123)
    $rs_time := 20000
    call release_all
  end if
  if ($CC_NUM = 1)
    $dyn := %CC[1]
    $dyn_knob := %CC[1]
    call calc_dyn
    call update_dyn
  end if
end on

on listener
  if ($NI_SIGNAL_TYPE = $NI_SIGNAL_TIMER_BEAT)
    inc($tick)
    if ($pq_n > 0)
      call process_pending
    end if
    if ($repeat > 0 and $rep_n > 0)
      $rp_k := 0
      while ($rp_k < 128)
        if (%rep_vel[$rp_k] > 0 and $tick >= %rep_next[$rp_k])
          $dh_note := $rp_k
          $dh_vel := %rep_vel[$rp_k]
          call drum_hit
          %rep_next[$rp_k] := $tick + %rep_ticks[$repeat]
        end if
        inc($rp_k)
      end while
    end if
  end if
end on
