function show_page
  $tab_play := 0
  $tab_sound := 0
  $tab_perf := 0
  if ($page = 0)
    $tab_play := 1
  end if
  if ($page = 1)
    $tab_sound := 1
  end if
  if ($page = 2)
    $tab_perf := 1
  end if
  $ui_i := 0
  while ($ui_i < $NUM_UI)
    if (%ui_page[$ui_i] = -1 or %ui_page[$ui_i] = $page)
      set_control_par(%ui_id[$ui_i], $CONTROL_PAR_HIDE, $HIDE_PART_NOTHING)
    else
      set_control_par(%ui_id[$ui_i], $CONTROL_PAR_HIDE, $HIDE_WHOLE_CONTROL)
    end if
    inc($ui_i)
  end while
end function

function play_voice
  $pv_id := play_note($pv_note, 127, $pv_off, 0)
  set_event_par_arr($pv_id, $EVENT_PAR_ALLOW_GROUP, 0, $ALL_GROUPS)
  set_event_par_arr($pv_id, $EVENT_PAR_ALLOW_GROUP, 1, $pv_grp)
  change_vol($pv_id, $pv_vol, 0)
  change_tune($pv_id, $pv_tune + $bend, 0)
  if ($pv_pan # 0)
    change_pan($pv_id, $pv_pan, 0)
  end if
  if ($pv_fade > 0)
    fade_in($pv_id, $pv_fade)
  end if
  if ($nv_n < 32)
    %nv_id[$nv_n] := $pv_id
    %nv_r[$nv_n] := $pv_role
    %nv_b[$nv_n] := $pv_vol
    inc($nv_n)
  end if
end function

function reg_new
  $rn_i := 0
  while ($rn_i < $nv_n)
    if (%src_n[$rs_src] < $MAXV)
      $rn_j := $rs_src * $MAXV + %src_n[$rs_src]
      %src_v[$rn_j] := %nv_id[$rn_i]
      %src_r[$rn_j] := %nv_r[$rn_i]
      %src_b[$rn_j] := %nv_b[$rn_i]
      %src_n[$rs_src] := %src_n[$rs_src] + 1
    else
      fade_out(%nv_id[$rn_i], 30000, 1)
    end if
    inc($rn_i)
  end while
end function

function release_src
  $rl_i := 0
  while ($rl_i < %src_n[$rs_src])
    fade_out(%src_v[$rs_src * $MAXV + $rl_i], $rs_time, 1)
    inc($rl_i)
  end while
  %src_n[$rs_src] := 0
  %src_sus[$rs_src] := 0
end function

function pend_new
  $pn_i := 0
  while ($pn_i < $nv_n)
    if ($pq_n < 512)
      %pq_id[$pq_n] := %nv_id[$pn_i]
      %pq_t[$pq_n] := $pq_rel
      inc($pq_n)
    else
      fade_out(%nv_id[$pn_i], 20000, 1)
    end if
    inc($pn_i)
  end while
end function

function process_pending
  $pp_i := 0
  while ($pp_i < $pq_n)
    if (%pq_t[$pp_i] <= $tick)
      fade_out(%pq_id[$pp_i], $rel_us, 1)
      dec($pq_n)
      %pq_id[$pp_i] := %pq_id[$pq_n]
      %pq_t[$pp_i] := %pq_t[$pq_n]
    else
      inc($pp_i)
    end if
  end while
end function

function release_all
  $ra_s := 0
  while ($ra_s < 129)
    if (%src_n[$ra_s] > 0)
      $rs_src := $ra_s
      call release_src
    end if
    %src_sus[$ra_s] := 0
    inc($ra_s)
  end while
  $ra_i := 0
  while ($ra_i < $pq_n)
    fade_out(%pq_id[$ra_i], $rs_time, 1)
    inc($ra_i)
  end while
  $pq_n := 0
end function

function pedal_up
  $pu_s := 0
  while ($pu_s < 128)
    if (%src_sus[$pu_s] = 1 and %src_held[$pu_s] = 0)
      $rs_src := $pu_s
      $rs_time := $rel_us
      call release_src
    end if
    inc($pu_s)
  end while
end function

    $fx_v := 0
  end if
  if ($fx_v > 1000000)
    $fx_v := 1000000
  end if
  set_engine_par($ENGINE_PAR_CUTOFF, $fx_v, -1, 0, $NI_INSERT_BUS)
end function

function scale_snap
  if ($scale_type > 0)
    $sc_pc := ($sc_note - $scale_key + 120) mod 12
    $sc_d := 0
    $sc_t := $sc_pc
    while (%scale_mask[$scale_type * 12 + $sc_t] = 0 and $sc_d < 11)
      inc($sc_d)
      $sc_t := ($sc_pc - $sc_d + 12) mod 12
    end while
    $sc_note := $sc_note - $sc_d
    if ($sc_note < 0)
      $sc_note := $sc_note + 12
    end if
  end if
end function

function build_chord
  $bc_n := 0
  %ch_note[$bc_src * 8] := $bc_root
  $bc_n := 1
  if ($chord = 15 or $chord = 16)
    $bc_need := 3
    if ($chord = 16)
      $bc_need := 4
    end if
    $sc_t := $scale_type
    if ($sc_t = 0)
      $sc_t := 1
    end if
    $bc_step := 0
    $bc_p := $bc_root
    while ($bc_n < $bc_need and $bc_p < $bc_root + 24)
      inc($bc_p)
      if (%scale_mask[$sc_t * 12 + (($bc_p - $scale_key + 120) mod 12)] = 1)
        inc($bc_step)
        if ($bc_step = 2 or $bc_step = 4 or $bc_step = 6)
          if ($bc_p <= 127)
            %ch_note[$bc_src * 8 + $bc_n] := $bc_p
            inc($bc_n)
          end if
        end if
      end if
    end while
  else
    $bc_i := 1
    while ($bc_i < %chord_n[$chord])
      $bc_p := $bc_root + %chord_iv[$chord * 6 + $bc_i]
      if ($bc_p <= 127)
        %ch_note[$bc_src * 8 + $bc_n] := $bc_p
        inc($bc_n)
      end if
      inc($bc_i)
    end while
  end if
  %ch_n[$bc_src] := $bc_n
end function

function arp_sort

    if ($as_i < $arp_n)
      %arp_sorted[$as_i] := %arp_note[$as_i]
    else
      %arp_sorted[$as_i] := 1000
    end if
    inc($as_i)
  end while
  sort(%arp_sorted, 0)
end function

function vel_curve
  ~vc_r := real($vc_in) / 127.0
  ~vc_r := pow(~vc_r, pow(2.0, real(0 - $vcurve) / 50.0))
  $vc_curved := int(~vc_r * 127.0)
  if ($vc_curved < 1)
    $vc_curved := 1
  end if
  $vc_db := int(real($vrange) * 1000.0 * (~vc_r - 1.0))
end function

function voice_vol
  $vr_m := $vr_role / 10
  $vr_r := $vr_role mod 10
  select ($vr_r)
    case 0
      $vr_vol := $dyn_gp + $dyn_db + $expr_db
    case 1
      $vr_vol := $dyn_gf + $dyn_db + $expr_db
    case 2
      $vr_vol := $dyn_db + $expr_db
    case 3
      $vr_vol := $vr_base + $expr_db
  end select
  $vr_vol := $vr_vol + %mic_db[$vr_m]
end function

function calc_dyn
  ~fr := real($dyn) / 127.0
  ~fg := cos(~fr * 1.5707963)
  if (~fg < 0.0001)
    ~fg := 0.0001
  end if
  $dyn_gp := int(20000.0 * log10(~fg))
  ~fg := sin(~fr * 1.5707963)
  if (~fg < 0.0001)
    ~fg := 0.0001
  end if
  $dyn_gf := int(20000.0 * log10(~fg))
  $dyn_db := 0 - int(real($dyn_range) * 1000.0 * pow(1.0 - ~fr, 1.3))
  ~fg := real($expr_cc) / 127.0
  if (~fg < 0.001)
    ~fg := 0.001
  end if
  $expr_db := int(20000.0 * log10(~fg) * 0.9)
end function

function start_note
  $nv_n := 0
  $pv_note := $sn_note
  $pv_tune := 0
  $pv_pan := 0
  $pv_off := $sn_off
  $pv_fade := $att_us
  if ($sn_fade >= 0)
    $pv_fade := $sn_fade
  end if
  if (%art_type[$art] = 1)
    $st_k := %rr_pos[$art]
    %rr_pos[$art] := (%rr_pos[$art] + 1) mod %art_gn[$art]
    $st_g := %grp[%art_g[$art * 4 + $st_k]]
    if ($st_g # $NI_NOT_FOUND)
      $vc_in := $sn_vel
      call vel_curve
      $st_base := $vc_db + random(-600, 600)
      $vr_role := 3
      $vr_base := $st_base
      call voice_vol
      $pv_grp := $st_g
      $pv_role := 3
      $pv_vol := $vr_vol
      $pv_fade := 0
      call play_voice
      %nv_b[$nv_n - 1] := $st_base
    end if
  else
    if (%art_type[$art] = 2)
      $pv_fade := $swell_t * 1000
    end if
    $st_k := 0
    while ($st_k < %art_gn[$art])
      $st_gi := %art_g[$art * 4 + $st_k]
      $st_g := %grp[$st_gi]
      if ($st_g # $NI_NOT_FOUND)
        if (%art_gn[$art] = 1)
          $vr_role := 2
        else
          $vr_role := $st_k
        end if
        $st_base := 0
        $st_role := $vr_role
        if ($dyn_src = 1)
          $st_v := $dyn
          $dyn := $sn_vel
          call calc_dyn
          call voice_vol
          $dyn := $st_v
          call calc_dyn
          $st_base := $vr_vol - $expr_db - %mic_db[0]
          $st_role := 3
        end if
        $st_m := 0
        while ($st_m < 2)
          $st_gg := -1
          if ($st_m = 0)
            $st_gg := $st_g
          else
            if (%hall_of[$st_gi] >= 0)
              $st_gg := %grp[%hall_of[$st_gi]]
            end if
          end if
          if ($st_gg # -1 and $st_gg # $NI_NOT_FOUND and ($st_m = 0))
            $pv_grp := $st_gg
            $pv_role := $st_role + 10 * $st_m
            $vr_role := $pv_role
            $vr_base := $st_base
            call voice_vol
            $pv_vol := $vr_vol
            call play_voice
            %nv_b[$nv_n - 1] := $st_base
          end if
          inc($st_m)
        end while
      end if
      inc($st_k)
    end while
  end if
end function

function arp_fire
  $af_total := $arp_n * $arp_oct
  if ($af_total > 0)
    $af_len := %rate_ticks[$arp_rate]
    $af_gate := ($af_len * $arp_gate) / 100
    if ($af_gate < 1)
      $af_gate := 1
    end if
    $af_vel := $arp_lastvel
    if (($tick mod 24) < $af_len)
      $af_vel := $af_vel + $arp_accent
    end if
    if ($af_vel > 127)
      $af_vel := 127
    end if
    if ($arp_mode = 6)
      $af_i := 0
      while ($af_i < $arp_n)
        $sn_note := %arp_note[$af_i]
        $sn_vel := $af_vel
        $sn_off := 0
        $sn_fade := -1
        call start_note
        $pq_rel := $tick + $af_gate
        call pend_new
        inc($af_i)
      end while
    else
      select ($arp_mode)
        case 1
          $af_idx := $arp_step mod $af_total
        case 2
          $af_idx := $af_total - 1 - ($arp_step mod $af_total)
        case 3
          if ($af_total = 1)
            $af_idx := 0
          else
            $af_per := 2 * $af_total - 2
            $af_idx := $arp_step mod $af_per
            if ($af_idx >= $af_total)
              $af_idx := $af_per - $af_idx
            end if
          end if
        case 4
          $af_idx := random(0, $af_total - 1)
        case 5
          $af_idx := $arp_step mod $af_total
      end select
      $af_e := $af_idx mod $arp_n
      $af_o := $af_idx / $arp_n
      if ($arp_mode = 5)
        $af_note := %arp_note[$af_e] + 12 * $af_o
      else
        $af_note := %arp_sorted[$af_e] + 12 * $af_o
      end if
      if ($af_note <= 127)
        $sn_note := $af_note
        $sn_vel := $af_vel
        $sn_off := 0
        $sn_fade := -1
        call start_note
        $pq_rel := $tick + $af_gate
        call pend_new
      end if
    end if
    inc($arp_step)
  end if
end function

function arp_add
  if ($arp_latch = 1 and $keys_down <= 1)
    $arp_n := 0
  end if
  $aa_was := $arp_n
  $aa_i := 0
  while ($aa_i < %ch_n[$aa_src])
    if ($arp_n < 64)
      %arp_note[$arp_n] := %ch_note[$aa_src * 8 + $aa_i]
      %arp_src[$arp_n] := $aa_src
      inc($arp_n)
    end if
    inc($aa_i)
  end while
  call arp_sort
  if ($aa_was = 0)
    $arp_step := 0
    call arp_fire
    $arp_next := $tick + %rate_ticks[$arp_rate]
  end if
end function

function arp_remove
  $ar_j := 0
  $ar_i := 0
  while ($ar_i < $arp_n)
    if (%arp_src[$ar_i] # $ar_src)
      %arp_note[$ar_j] := %arp_note[$ar_i]
      %arp_src[$ar_j] := %arp_src[$ar_i]
      inc($ar_j)
    end if
    inc($ar_i)
  end while
  $arp_n := $ar_j
  call arp_sort
end function

function update_dyn
  $ud_s := 0
  while ($ud_s < 128)
    if (%src_n[$ud_s] > 0)
      $ud_i := 0
      while ($ud_i < %src_n[$ud_s])
        $ud_j := $ud_s * $MAXV + $ud_i
        $vr_role := %src_r[$ud_j]
        $vr_base := %src_b[$ud_j]
        call voice_vol
        change_vol(%src_v[$ud_j], $vr_vol, 0)
        inc($ud_i)
      end while
    end if
    inc($ud_s)
  end while
  $ud_i := 0
  while ($ud_i < $mono_n)
    $vr_role := %mono_r[$ud_i]
    $vr_base := %mono_b[$ud_i]
    call voice_vol
    change_vol(%mono_v[$ud_i], $vr_vol, 0)
    inc($ud_i)
  end while
end function

function stack_remove
  $sk_j := 0
  $sk_i := 0
  while ($sk_i < $stk_n)
    if (%stk[$sk_i] # $sk_note)
      %stk[$sk_j] := %stk[$sk_i]
      inc($sk_j)
    end if
    inc($sk_i)
  end while
  $stk_n := $sk_j
end function

on note
  ignore_event($EVENT_ID)
  if (%art_ks[0] >= 0)
    $ks_i := 0
    while ($ks_i < $NUM_ART)
      if ($EVENT_NOTE = %art_ks[$ks_i])
        $art := $ks_i
        $art_menu := $ks_i
        call color_keys
        exit
      end if
      inc($ks_i)
    end while
  end if
  inc($keys_down)
  %src_held[$EVENT_NOTE] := 1
  if ($legato = 1 and %art_type[$art] # 1)
    $mono_vel := $EVENT_VELOCITY
    $sk_note := $EVENT_NOTE
    call stack_push
    inc($gl_token)
    $p_tok := $gl_token
    if ($mono_n > 0)
      $p_from := ($mono_cur - $EVENT_NOTE) * 1000 * $porta
      call mono_fade_old
      $sn_note := $EVENT_NOTE
      $sn_vel := $mono_vel
      $sn_off := 60000
      $sn_fade := $leg_time * 1000
      call start_note
      call mono_take
      $mono_cur := $EVENT_NOTE
      if ($p_from # 0)
        $p_steps := $leg_time / 5
        if ($p_steps < 1)
          $p_steps := 1
        end if
        $p_s := 0
        while ($p_s <= $p_steps and $p_tok = $gl_token)
          $mt_i := 0
          while ($mt_i < $mono_n)
            change_tune(%mono_v[$mt_i], $p_from - ($p_from / $p_steps) * $p_s + $bend, 0)
            inc($mt_i)
          end while
          wait(5000)
          inc($p_s)
        end while
      end if
    else
      $sn_note := $EVENT_NOTE
      $sn_vel := $EVENT_VELOCITY
      $sn_off := 0
      $sn_fade := -1
      call start_note
      call mono_take
      $mono_cur := $EVENT_NOTE
    end if
    exit
  end if
  $p_vel := $EVENT_VELOCITY
  if ($hum_vel > 0)
    $p_vel := $p_vel + random(0 - $hum_vel, $hum_vel)
    if ($p_vel < 1)
      $p_vel := 1
    end if
    if ($p_vel > 127)
      $p_vel := 127
    end if
  end if
  $sc_note := $EVENT_NOTE
  call scale_snap
  $bc_root := $sc_note
  $bc_src := $EVENT_NOTE
  call build_chord
  if ($arp_mode > 0)
    $arp_lastvel := $p_vel
    $aa_src := $EVENT_NOTE
    call arp_add
    exit
  end if
  if (%src_n[$EVENT_NOTE] > 0)
    $rs_src := $EVENT_NOTE
    $rs_time := $retrig_us
    call release_src
  end if
  if ($hum_time > 0)
    wait(random(0, $hum_time * 1000) + 1)
  end if
  $p_i := 0
  $p_cnt := %ch_n[$EVENT_NOTE]
  while ($p_i < $p_cnt)
    if ($p_i > 0 and $strum > 0)
      wait($strum * 1000)
    end if
    if (%src_held[$EVENT_NOTE] = 1 or $pedal = 1)
      $sn_note := %ch_note[$EVENT_NOTE * 8 + $p_i]
      $sn_vel := $p_vel
      $sn_off := 0
      $sn_fade := -1
      call start_note
      $rs_src := $EVENT_NOTE
      call reg_new
    else
      $p_i := $p_cnt
    end if
    inc($p_i)
  end while
end on

on release
  $ks_i := 0
  while ($ks_i < $NUM_ART)
    if ($EVENT_NOTE = %art_ks[$ks_i])
      exit
    end if
    inc($ks_i)
  end while
  dec($keys_down)
  if ($keys_down < 0)
    $keys_down := 0
  end if
  %src_held[$EVENT_NOTE] := 0
  if ($legato = 1 and $mono_n > 0)
    $sk_note := $EVENT_NOTE
    call stack_remove
    if ($EVENT_NOTE = $mono_cur)
      if ($stk_n > 0)
        $q_tgt := %stk[$stk_n - 1]
        inc($gl_token)
        $q_tok := $gl_token
        if ($mono_n > 0)
          $q_from := ($mono_cur - $q_tgt) * 1000 * $porta
          call mono_fade_old
          $sn_note := $q_tgt
          $sn_vel := $mono_vel
          $sn_off := 60000
          $sn_fade := $leg_time * 1000
          call start_note
          call mono_take
          $mono_cur := $q_tgt
          if ($q_from # 0)
            $q_steps := $leg_time / 5
            if ($q_steps < 1)
              $q_steps := 1
            end if
            $q_s := 0
            while ($q_s <= $q_steps and $q_tok = $gl_token)
              $mt_i := 0
              while ($mt_i < $mono_n)
                change_tune(%mono_v[$mt_i], $q_from - ($q_from / $q_steps) * $q_s + $bend, 0)
                inc($mt_i)
              end while
              wait(5000)
              inc($q_s)
            end while
          end if
        else
          $sn_note := $q_tgt
          $sn_vel := $mono_vel
          $sn_off := 0
          $sn_fade := -1
          call start_note
          call mono_take
          $mono_cur := $q_tgt
        end if
      else
        $mt_i := 0
        while ($mt_i < $mono_n)
          fade_out(%mono_v[$mt_i], $rel_us, 1)
          inc($mt_i)
        end while
        $mono_n := 0
        inc($gl_token)
      end if
    end if
    exit
  end if
  if ($arp_mode > 0)
    if ($arp_latch = 0)
      $ar_src := $EVENT_NOTE
      call arp_remove
    end if
    exit
  end if
  if ($pedal = 1)
    %src_sus[$EVENT_NOTE] := 1
  else
    $rs_src := $EVENT_NOTE
    $rs_time := $rel_us
    call release_src
  end if
end on

on controller
  if ($CC_NUM = 64 and $use_pedal = 1)
    if (%CC[64] >= 64)
      $pedal := 1
    else
      if ($pedal = 1)
        $pedal := 0
        call pedal_up
      end if
    end if
  end if
  if ($CC_NUM = 120 or $CC_NUM = 123)
    $rs_time := 20000
    call release_all
  end if
  if ($CC_NUM = $VCC_PITCH_BEND)
    $cb_v := (%CC[$VCC_PITCH_BEND] * $pb_range * 100) / 8191 * 1000
    change_tune($ALL_EVENTS, $cb_v - $bend, 1)
    $bend := $cb_v
  end if
  if ($CC_NUM = 1 and $dyn_src = 0)
    $dyn := %CC[1]
    $dyn_knob := %CC[1]
    call calc_dyn
    call update_dyn
  end if
  if ($CC_NUM = 11)
    $expr_cc := %CC[11]
    call calc_dyn
    call update_dyn
  end if
end on

on listener
  if ($NI_SIGNAL_TYPE = $NI_SIGNAL_TIMER_BEAT)
    inc($tick)
    if ($pq_n > 0)
      call process_pending
    end if
    if ($arp_mode > 0 and $arp_n > 0 and $tick >= $arp_next)
      call arp_fire
      $lt_len := %rate_ticks[$arp_rate]
      $lt_sw := ($lt_len * $arp_swing) / 6
      if (($arp_step mod 2) = 1)
        $arp_next := $tick + $lt_len + $lt_sw
      else
        $arp_next := $tick + $lt_len - $lt_sw
      end if
    end if
  end if
end on
